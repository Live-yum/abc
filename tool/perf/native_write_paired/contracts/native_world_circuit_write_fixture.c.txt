/* Host WRITE/RESULT lifetime contract. Every ack overwrites the same C buffer.
 * WRITE uses nonsequential offsets; RESULT must survive reuse and final ack. */
#include <stdint.h>
#include <string.h>

static uint8_t payload[64];
static uint32_t metrics[12];
static uint32_t command_kind, index_in_command, pending, invalid;
static const uint32_t offsets[] = {11, 0, 29};
static const uint32_t lengths[] = {7, 17, 9};

void write_test_configure(uint32_t invalid_event) {
  memset(metrics, 0, sizeof(metrics));
  memset(payload, 0xda, sizeof(payload));
  command_kind = index_in_command = pending = 0;
  invalid = invalid_event;
}
uint32_t write_test_metric(uint32_t index) { return metrics[index]; }
uint32_t write_test_pending(void) { return pending; }
void write_test_mutate(uint32_t index, uint32_t value) {
  if (index < sizeof(payload)) payload[index] = (uint8_t)value;
}
const char *abc_engine_build_info(void) {
  return "{\"circuitWorldAbiVersion\":2}";
}
int32_t abc_world_open(const uint8_t *bytes, uint32_t size, uint32_t *out) {
  if (metrics[7]) return -90;
  metrics[7] = 1; *out = 1; return 0;
}
int32_t abc_world_stream_open_begin(uint32_t source, uint32_t size,
    uint32_t *out) { *out = 11; return 0; }
int32_t abc_world_stream_step(uint32_t task, uint32_t budget,
    uint32_t *event, uint8_t **data) {
  memset(event, 0, 12 * sizeof(uint32_t));
  event[0] = 1; event[1] = 4; *data = 0; return 0;
}
int32_t abc_world_stream_adopt(uint32_t task, uint32_t source, uint32_t *out) {
  return abc_world_open(0, 0, out);
}
int32_t abc_world_stream_close(uint32_t task) { return 0; }
int32_t abc_world_stream_cancel(uint32_t task) { return 0; }
int32_t abc_world_circuit_begin(uint32_t world, uint32_t scratch,
    uint32_t budget, uint32_t *out) {
  if (!metrics[7] || metrics[8]) return -91;
  metrics[8] = 1; command_kind = index_in_command = pending = 0;
  *out = 7; return 0;
}
int32_t abc_world_circuit_command(uint32_t id, const uint32_t *words,
    const uint32_t *records) {
  if (!metrics[8] || pending) { ++metrics[9]; return -92; }
  ++metrics[10]; command_kind = words[1]; index_in_command = 0; return 0;
}
int32_t abc_world_circuit_step(uint32_t id, uint32_t budget,
    uint32_t *event, uint8_t **data) {
  ++metrics[0];
  if (!metrics[8] || pending) { ++metrics[9]; return -93; }
  memset(event, 0, 12 * sizeof(uint32_t)); event[0] = 2; *data = 0;
  const int writes = command_kind == 0 || command_kind == 6;
  const uint32_t count = writes ? 3 : 2;
  if (index_in_command >= count) {
    event[1] = 4; event[9] = command_kind;
    event[10] = command_kind == 6 ? 38 : writes ? 0 : 2;
    return 0;
  }
  for (uint32_t i = 0; i < sizeof(payload); ++i)
    payload[i] = (uint8_t)(index_in_command * 37u + i + 3u);
  event[1] = pending = writes ? 2 : 3;
  event[2] = command_kind == 6 ? 3 : 2;
  event[3] = writes ? offsets[index_in_command] : 0;
  event[4] = writes ? lengths[index_in_command] : 16;
  event[9] = command_kind; event[10] = 1; *data = payload;
  ++metrics[writes ? 1 : 2];
  if (invalid == 1) event[2] = 99;
  if (invalid == 2) *data = 0;
  if (invalid == 3) event[4] = 1024 * 1024 + 1;
  if (invalid == 4) event[3] = UINT32_MAX;
  return 0;
}
int32_t abc_world_circuit_ack(uint32_t id) {
  if (!pending) { ++metrics[9]; return -94; }
  ++metrics[3]; if (pending == 2) ++metrics[11];
  memset(payload, 0xda, sizeof(payload));
  pending = 0; ++index_in_command; return 0;
}
int32_t abc_world_circuit_stats(uint32_t id, uint32_t *stats) {
  memset(stats, 0, 24 * sizeof(uint32_t)); stats[0] = 2; return 0;
}
int32_t abc_world_circuit_cancel(uint32_t id) {
  ++metrics[4]; pending = 0; memset(payload, 0xda, sizeof(payload)); return 0;
}
int32_t abc_world_circuit_close(uint32_t id) {
  ++metrics[5];
  if (!metrics[8]) { ++metrics[9]; return -95; }
  metrics[8] = pending = 0; memset(payload, 0xda, sizeof(payload)); return 0;
}
int32_t abc_world_close(uint32_t world) {
  ++metrics[6];
  if (metrics[8] || !metrics[7]) { ++metrics[9]; return -96; }
  metrics[7] = 0; return 0;
}
