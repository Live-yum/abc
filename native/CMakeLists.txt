cmake_minimum_required(VERSION 3.27)
project(abc_engine C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ABC_TERRA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/TerraWasm" CACHE PATH "Authorized pointer-safe TerraWasm source")
option(ABC_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(NOT EXISTS "${ABC_TERRA_SOURCE}/include/terra_world.h")
 message(FATAL_ERROR "Supply authorized TerraWasm sources using ABC_TERRA_SOURCE")
endif()
set(TERRAWASM_NATIVE_TESTS OFF CACHE BOOL "" FORCE)
set(TERRAWASM_FEATURE_SET all CACHE STRING "" FORCE)
set(TERRAWASM_SANITIZERS ${ABC_SANITIZERS} CACHE BOOL "" FORCE)
add_subdirectory("${ABC_TERRA_SOURCE}" terra)
set_target_properties(terrax_world_static PROPERTIES C_VISIBILITY_PRESET hidden)
if(ANDROID)
 set(ABC_ZLIB z)
else()
 find_package(ZLIB REQUIRED)
 set(ABC_ZLIB ZLIB::ZLIB)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
 add_library(abc_engine STATIC abc_engine.c abc_circuit.c abc_circuit_rules.c abc_region.c abc_world_circuit.c)
else()
 add_library(abc_engine SHARED abc_engine.c abc_circuit.c abc_circuit_rules.c abc_region.c abc_world_circuit.c)
endif()
target_include_directories(abc_engine PRIVATE "${ABC_TERRA_SOURCE}/include" "${ABC_TERRA_SOURCE}/src")
target_link_libraries(abc_engine PRIVATE terrax_world_static ${ABC_ZLIB})
if(UNIX)
 target_link_libraries(abc_engine PRIVATE m)
endif()
if(ABC_SANITIZERS)
 target_compile_options(abc_engine PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(abc_engine PRIVATE -fsanitize=address,undefined)
endif()
